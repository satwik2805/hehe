import os
import shutil

try:
    hdfs_root = "HDFS"
    os.makedirs(hdfs_root, exist_ok=True)
    print("Root HDFS directory created")

    input_dir = os.path.join(hdfs_root, "input")
    output_dir = os.path.join(hdfs_root, "output")
    backup_dir = os.path.join(hdfs_root, "backup")

    os.makedirs(input_dir, exist_ok=True)
    os.makedirs(output_dir, exist_ok=True)
    os.makedirs(backup_dir, exist_ok=True)
    print("Subdirectories created")

    file_path = os.path.join(input_dir, "data.txt")
    with open(file_path, "w") as file:
        file.write("Welcome to HDFS")
    print("File created")

    copied_file = os.path.join(backup_dir, "data.txt")
    shutil.copy(file_path, copied_file)
    print("File copied")

    moved_file = os.path.join(output_dir, "data.txt")
    shutil.move(file_path, moved_file)
    print("File moved")

    renamed_file = os.path.join(output_dir, "data_final.txt")
    os.rename(moved_file, renamed_file)
    print("File renamed")

    print("\nHDFS directory structure")
    for root, dirs, files in os.walk(hdfs_root):
        print(f"\n{root}")
        for d in dirs:
            print("DIR:", d)
        for f in files:
            print("FILE:", f)

    os.remove(copied_file)
    print("Copied file deleted")

    os.rmdir(backup_dir)
    print("Empty backup directory deleted")

    shutil.rmtree(output_dir)
    print("Non-empty output directory deleted")

except FileNotFoundError:
    print("Error: File or directory not found")
except PermissionError:
    print("Error: Permission denied")
except Exception as e:
    print("An error occurred:", e)

import json

file_name = "students.json"


def load_data():
    try:
        with open(file_name, "r") as file:
            return json.load(file)
    except:
        return []


def save_data(data):
    with open(file_name, "w") as file:
        json.dump(data, file, indent=4)


def create_record():
    data = load_data()
    sid = input("Enter student ID: ")
    name = input("Enter student name: ")
    age = int(input("Enter age: "))
    course = input("Enter course: ")

    student = {
        "ID": sid,
        "Name": name,
        "Age": age,
        "Course": course
    }

    data.append(student)
    save_data(data)
    print("Record added successfully")


def read_records():
    data = load_data()
    if len(data) == 0:
        print("No records found")
    else:
        for student in data:
            print(student)


def update_record():
    data = load_data()
    sid = input("Enter student ID to update: ")
    for student in data:
        if student["ID"] == sid:
            student["Name"] = input("Enter New Name: ")
            student["Age"] = int(input("Enter New Age: "))
            student["Course"] = input("Enter New Course: ")
            save_data(data)
            print("Record updated successfully")
            return
    print("Record not found")


def delete_record():
    data = load_data()
    sid = input("Enter student ID to delete: ")
    for student in data:
        if student["ID"] == sid:
            data.remove(student)
            save_data(data)
            print("Record deleted")
            return
    print("Record not found")


while True:
    print("\n--- JSON CRUD OPERATIONS ---")
    print("1. Create record")
    print("2. Read records")
    print("3. Update record")
    print("4. Delete record")
    print("5. Exit")

    choice = input("Enter your choice: ")

    if choice == "1":
        create_record()
    elif choice == "2":
        read_records()
    elif choice == "3":
        update_record()
    elif choice == "4":
        delete_record()
    elif choice == "5":
        print("Exiting program")
        break
    else:
        print("Invalid choice")

import pandas as pd

    d = pd.DataFrame({
        "user_id": [101, 102, 103, 105, 106, 107],
        "action": ["login", "login", "login", "purchase", "logout", "login"],
        "timestamp": ["10:00", "10:13", "10:20", "10:30", "10:45", "11:00"]

    print("\nHIVE Analysis\n", d)
    print("\nSelect action, count(*)\n", d.groupby("action").size())
    print("\nGroup by action\n", d.groupby("action").size().to_string(index=False))

    print("\nPig Analysis\n\nLog Data:\n", d)
    e = d[d["type"] == "error"]
    print("\nFiltered Logs:\n", e)

from pyspark.sql import SparkSession
from pyspark.sql.functions import avg, max, min, count

s = SparkSession.builder.appName("StudentDataAnalysis").getOrCreate()
d = [
    (101, "Rahul", "CSE", 85), (102, "Pooja", "ISE", 90),
    (103, "Arun", "CSE", 78), (104, "Sneha", "ECE", 88),
    (105, "Kiran", "ISE", 92), (106, "Anu", "ECE", 75)
]
df = s.createDataFrame(d, ["ID", "Name", "Department", "Marks"])

print("Student Data:")
df.show()
print("Students with marks > 80")
df.filter(df.Marks > 80).show()
print("Average marks by department")
df.groupBy("Department").agg(avg("Marks").alias("Average Marks")).show()
print("Maximum Marks")
df.agg(max("Marks").alias("Maximum Marks")).show()
print("Minimum Marks")
df.agg(min("Marks").alias("Minimum Marks")).show()
print("Number of students in each department")
df.groupBy("Department").agg(count("*").alias("Student Count")).show()
s.stop()
